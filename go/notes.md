CLI
go mod(go modules) - менеджер пакетов

go mod init       # название/ссылка на репозиторий - инициализирует модуль
go mod tidy       # смотрит проект и устанавливает используемые пакеты и удаляет неиспользумеые из go.mod
go mod download   # скачать все зависимости из go.mod (без сборки) CI/CD, хочешь скачать отдельно от сборки
go mod verify     # проверить контрольные суммы (go.sum) CI/CD, проверка безопасности перед деплоем
go mod vendor     # скопировать зависимости в папку vendor/ нет интернета на сервере, корпоративные ограничения

go get github.com/gin-gonic/gin          # установить последнюю версию
go get github.com/gin-gonic/gin@v1.9.1   # конкретная версия
go get github.com/gin-gonic/gin@latest   # явно последнюю
go get github.com/gin-gonic/gin@none     # удалить пакет

go.mod - файл с зависимостями/аналог package.json 
go.sum - файл с контрольными суммами пакетов/аналог package-lock.json
Оба файла коммитятся!

Типы данных

Целые числа

var i int = 42

// Знаковые (могут быть отрицательными)
var i8  int8  = 127          // от -128 до 127
var i16 int16 = 32767        // от -32768 до 32767
var i32 int32 = 2147483647   // от -2млрд до 2млрд
var i64 int64 = 9223372036854775807

// Беззнаковые (только положительные)
var u8  uint8  = 255         // от 0 до 255
var u16 uint16 = 65535
var u32 uint32 = 4294967295
var u64 uint64 = 18446744073709551615

С плавающей запятой
var f32 float32 = 3.14    // ~6-7 значимых цифр
var f64 float64 = 3.14    // ~15-16 значимых цифр
float64 используй всегда — float32 только если критична память.
