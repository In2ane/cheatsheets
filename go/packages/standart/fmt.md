Классификаторы:
```go
fmt.Printf("%d", 42)      // 42        — целое десятичное
fmt.Printf("%b", 42)      // 101010    — двоичное
fmt.Printf("%o", 42)      // 52        — восьмеричное
fmt.Printf("%x", 42)      // 2a        — шестнадцатеричное (строчные)
fmt.Printf("%X", 42)      // 2A        — шестнадцатеричное (прописные)

fmt.Printf("%f", 3.14)    // 3.140000  — обычный
fmt.Printf("%e", 3.14)    // 3.140000e+00 — научный формат
fmt.Printf("%g", 3.14)    // 3.14      — компактный (сам выбирает)

fmt.Printf("%s", "привет") // привет   — строка
fmt.Printf("%q", "привет") // "привет" — строка в кавычках
fmt.Printf("%c", 65)       // A        — символ по коду

fmt.Printf("%v", что_угодно)  // значение в дефолтном формате
fmt.Printf("%T", что_угодно)  // тип переменной
fmt.Printf("%p", &x)          // адрес в памяти

fmt.Printf("%.2f", 3.14159) // 3.14     — 2 знака после запятой
fmt.Printf("%10d", 42)      //         42 — ширина 10
fmt.Printf("%#x", 42)       // 0x2a    — с префиксом системы счисления
fmt.Printf("%[1]d %[1]b", 42) // 42 101010 — повторное использование аргумента
```
